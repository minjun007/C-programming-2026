#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>

int main(void)
{
    int month;
    int days;

    printf("월 입력(1~12) : ");
    scanf("%d", &month);

    if (month >= 1 && month <= 12) {
        switch (month) {
            case 2:
                days = 28;
                break;
            case 4:
            case 6:
            case 9:
            case 11:
                days = 30;
                break;
            default:
                days = 31;
                break;
        }
        printf("%d월은 %d일 까지 있습니다.\n", month, days);
    } else {
        printf("1부터 12사이의 값을 입력하세요.\n");
    }

    return 0;
}
